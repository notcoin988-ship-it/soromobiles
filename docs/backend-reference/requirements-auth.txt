# FastAPI, SQLAlchemy и драйвер PostgreSQL здесь НЕ перечислены намеренно — они

# конфликтовать с тем, что там стоит. Ставится поверх:
#
#   pip install -r docs/backend-reference/requirements-auth.txt

# Подпись и проверка access-токенов (JWT).
PyJWT>=2.8,<3

# Проверка id_token из нативного окна Google (POST /v1/auth/google): сверяет
# подпись публичными ключами Google, срок годности и аудиторию. Ключи
# библиотека кеширует сама — похода в сеть на каждый вход нет.
google-auth>=2.30,<3

# ЧЕГО ЗДЕСЬ БОЛЬШЕ НЕТ И ПОЧЕМУ.
#
# passlib[bcrypt]==1.7.4 + bcrypt==4.0.1 — вход по паролю из мобильного
# контракта убран (вход только через Google), хешировать нечего. Заодно ушёл
# самый неприятный пункт развёртывания: passlib 1.7.4 несовместим с bcrypt
# >= 4.1 и падает при первом же хешировании, а на 5.x ещё и ValueError про
# 72 байта. Если веб-проект хешует пароли своим кодом — это его зависимости,
# закреплённые у него; предупреждение про верхнюю границу bcrypt остаётся
# верным и там.
#
# email-validator — EmailStr в схемах больше не встречается: почта приходит от
# Google уже проверенной, и валидировать её формой нечего.
